# P3 — Live Trading on Interactive Brokers (paper first) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Run the *same* `Strategy` live against Interactive Brokers (paper account first) through an event-driven `LiveEngine` + `IBBroker`, with hard mode-gating (DRY_RUN → PAPER → LIVE), the same `RiskManager` kill-switch, and persisted state.

**Architecture:** The `Broker` ABC gains live lifecycle methods (`connect`/`disconnect`/`cancel`/`positions`/`account_value`) and a tick-callback hook — all with safe defaults so `SimulatedBroker` is unchanged. `IBBroker` wraps an **injected** `ib_async.IB` client: it translates core `Order`→IB order, IB execution events→core `Fill`, IB tickers→core `Tick`. The injection makes every translation path unit-testable with a fake IB (no live connection). `LiveEngine` uses a `WallClock`, receives ticks/fills via callbacks, routes orders through the same `RiskManager`, gates submission by `TradingMode`, persists state to disk, and flattens on the kill switch. A `fxtrader live` command drives it; `LIVE` mode requires an explicit `--i-understand-live` flag. The real IB connection is exercised by an opt-in manual smoke test (needs IB Gateway running).

**Tech Stack:** Python 3.12 (`.venv`), `ib_async>=2` (installed; introspected API: `IB.connect/placeOrder/cancelOrder/positions`, events `execDetailsEvent(trade, fill)` / `pendingTickersEvent(tickers)`, `Forex`/`MarketOrder`/`LimitOrder`, `Execution.side` ∈ {BOT, SLD}). Reuses P2a `Strategy`/`RiskManager`/`Portfolio`. pytest with fakes — no live tests in the default suite.

**Conventions:**
- All timestamps tz-aware UTC. Use `.venv/bin/pytest` / `.venv/bin/python` (shell `source` doesn't persist). pytest uses `--import-mode=importlib`.
- `ib_async` is an OPTIONAL extra (`[live]`), also added to `[dev]` so tests import it. `broker/ib.py` may import `ib_async` at module top.
- Default-safe: `LiveEngine` defaults to `DRY_RUN` (no orders sent). `LIVE` is gated behind an explicit flag.
- Commit after each task. No placeholders.

```python
"""IBBroker — live execution via Interactive Brokers (ib_async).

The ib_async IB client is injected so every translation path is unit-testable with a
fake. Real connection is exercised by the opt-in integration smoke test.
"""

from __future__ import annotations

import math
from datetime import datetime, timezone

from ib_async import Forex, LimitOrder, MarketOrder

from fxtrader.broker.base import Broker
from fxtrader.core.types import Fill, Order, OrderType, Side, Tick


class IBBroker(Broker):
    def __init__(
        self,
        ib=None,
        *,
        host: str = "127.0.0.1",
        port: int = 7497,  # 7497 = TWS paper, 4002 = Gateway paper
        client_id: int = 1,
        symbols: list[str] | None = None,
    ) -> None:
        self._ib = ib
        self.host = host
        self.port = port
        self.client_id = client_id
        self.symbols = symbols or []
        self._last_tick: dict[str, Tick] = {}

    # --- connection / lifecycle ---
    def _client(self):
        if self._ib is None:
            from ib_async import IB
            self._ib = IB()
        return self._ib

    def connect(self) -> None:
        ib = self._client()
        ib.connect(self.host, self.port, clientId=self.client_id)
        ib.execDetailsEvent += self._on_exec_details
        ib.pendingTickersEvent += self._on_pending_tickers
        for symbol in self.symbols:
            ib.reqMktData(self._contract(symbol))

    def disconnect(self) -> None:
        if self._ib is not None and self._ib.isConnected():
            self._ib.disconnect()

    # --- helpers ---
    @staticmethod
    def _contract(symbol: str):
        return Forex(symbol)

    @staticmethod
    def _symbol_of(contract) -> str:
        return f"{contract.symbol}{contract.currency}"

    # --- order routing ---
    def submit(self, order: Order) -> None:
        action = "BUY" if order.side is Side.BUY else "SELL"
        if order.order_type is OrderType.MARKET:
            ib_order = MarketOrder(action, order.quantity)
        elif order.order_type is OrderType.LIMIT:
            ib_order = LimitOrder(action, order.quantity, order.limit_price)
        else:
            raise NotImplementedError(f"IBBroker does not support {order.order_type}")
        self._client().placeOrder(self._contract(order.symbol), ib_order)

    def cancel(self, order: Order) -> None:
        # Cancel any open IB orders for the symbol (best-effort; live order tracking is P3+).
        ib = self._client()
        for trade in ib.openTrades():
            if self._symbol_of(trade.contract) == order.symbol:
                ib.cancelOrder(trade.order)

    def update_market(self, tick: Tick) -> None:
        self._last_tick[tick.symbol] = tick

    def positions(self) -> dict[str, float]:
        out: dict[str, float] = {}
        for pos in self._client().positions():
            out[self._symbol_of(pos.contract)] = pos.position
        return out

    def account_value(self) -> float | None:
        for value in self._client().accountValues():
            if value.tag == "NetLiquidation":
                return float(value.value)
        return None

    # --- event handlers (translate IB -> core; unit-tested directly) ---
    def _on_exec_details(self, trade, fill) -> None:
        execution = fill.execution
        side = Side.BUY if execution.side == "BOT" else Side.SELL
        commission = 0.0
        report = getattr(fill, "commissionReport", None)
        if report is not None and getattr(report, "commission", None) is not None:
            commission = float(report.commission)
        self._emit_fill(
            Fill(
                order_id=str(execution.orderId),
                symbol=self._symbol_of(fill.contract),
                side=side,
                quantity=float(execution.shares),
                price=float(execution.price),
                ts=execution.time,
                commission=commission,
            )
        )

    def _on_pending_tickers(self, tickers) -> None:
        for ticker in tickers:
            bid, ask = ticker.bid, ticker.ask
            if bid is None or ask is None or math.isnan(bid) or math.isnan(ask):
                continue
            tick = Tick(
                symbol=self._symbol_of(ticker.contract),
                ts=ticker.time or datetime.now(timezone.utc),
                bid=float(bid),
                ask=float(ask),
                bid_volume=float(ticker.bidSize or 0.0),
                ask_volume=float(ticker.askSize or 0.0),
            )
            self._last_tick[tick.symbol] = tick
            self._emit_tick(tick)
```

```python
"""LiveEngine — real-time, callback-driven execution mirroring the backtest loop.

Same Strategy / RiskManager as the backtester. Mode-gated order submission
(DRY_RUN sends nothing). Kill-switch flattens via the broker. State persists to disk.
"""

from __future__ import annotations

import json
from enum import Enum
from pathlib import Path

from fxtrader.broker.base import Broker
from fxtrader.core.clock import WallClock
from fxtrader.core.portfolio import Portfolio
from fxtrader.core.types import Order, Side, Tick
from fxtrader.risk.manager import RiskManager
from fxtrader.strategy.base import Strategy


class TradingMode(str, Enum):
    DRY_RUN = "DRY_RUN"  # compute orders but send nothing to the broker
    PAPER = "PAPER"      # paper account
    LIVE = "LIVE"        # real money


class LiveEngine:
    def __init__(
        self,
        broker: Broker,
        strategy: Strategy,
        portfolio: Portfolio,
        risk: RiskManager,
        *,
        mode: TradingMode = TradingMode.DRY_RUN,
        state_path: str | Path | None = None,
    ) -> None:
        self.broker = broker
        self.strategy = strategy
        self.portfolio = portfolio
        self.risk = risk
        self.mode = mode
        self.state_path = Path(state_path) if state_path else None
        self.clock = WallClock()
        self._last_price: dict[str, float] = {}
        self._flattened = False

    def start(self) -> None:
        self.broker.set_fill_callback(self._on_fill)
        self.broker.set_tick_callback(self._on_tick)
        self.strategy.bind(self.clock, self.portfolio, self._on_order)
        self.broker.connect()
        self.strategy.on_start()

    def stop(self) -> None:
        self.strategy.on_stop()
        self._persist_state()
        self.broker.disconnect()

    def _on_order(self, order: Order) -> None:
        if self.risk.halted:
            return
        if order.symbol not in self._last_price:
            return
        if not self.risk.check(order, self.portfolio, self._last_price, self.clock):
            return
        if self.mode is TradingMode.DRY_RUN:
            return  # computed but not sent
        self.broker.submit(order)

    def _on_fill(self, fill) -> None:
        self.portfolio.apply_fill(fill)
        self.strategy.on_fill(fill)

    def _on_tick(self, tick: Tick) -> None:
        self._last_price[tick.symbol] = tick.mid
        self.broker.update_market(tick)

        equity = self.portfolio.equity(self._last_price)
        self.risk.update_equity(equity, self.clock)
        if self.risk.halted:
            if not self._flattened:
                self._flatten()
        else:
            self._flattened = False
            self.strategy.on_tick(tick)

        self._persist_state()

    def _flatten(self) -> None:
        for symbol, pos in list(self.portfolio.positions.items()):
            if pos.quantity != 0 and symbol in self._last_price:
                side = Side.SELL if pos.quantity > 0 else Side.BUY
                # Flatten orders bypass mode gating ONLY in non-dry-run; a kill switch
                # must be able to close real exposure. In DRY_RUN there is none.
                if self.mode is not TradingMode.DRY_RUN:
                    self.broker.submit(Order(symbol, side, abs(pos.quantity)))
        self._flattened = True

    def _persist_state(self) -> None:
        if self.state_path is None:
            return
        state = {
            "mode": self.mode.value,
            "ts": self.clock.now().isoformat(),
            "equity": self.portfolio.equity(self._last_price),
            "cash": self.portfolio.cash,
            "halted": self.risk.halted,
            "positions": {s: p.quantity for s, p in self.portfolio.positions.items() if p.quantity},
        }
        self.state_path.parent.mkdir(parents=True, exist_ok=True)
        tmp = self.state_path.with_suffix(self.state_path.suffix + ".tmp")
        tmp.write_text(json.dumps(state, indent=2))
        tmp.replace(self.state_path)
```

def _resolve_mode(mode: str, *, confirm_live: bool) -> TradingMode:
    key = mode.strip().upper().replace("-", "_")
    resolved = TradingMode(key)  # raises ValueError on a bad mode string
    if resolved is TradingMode.LIVE and not confirm_live:
        raise typer.BadParameter("LIVE trading requires --i-understand-live (real money).")
    return resolved

@app.command()
def live(
    symbol: str = typer.Option(..., help="Single symbol, e.g. EURUSD"),
    mode: str = typer.Option("dry-run", help="dry-run | paper | live"),
    fast: int = typer.Option(20, help="Fast MA window (bars)"),
    slow: int = typer.Option(50, help="Slow MA window (bars)"),
    size: float = typer.Option(10_000, help="Position size (base units)"),
    host: str = typer.Option("127.0.0.1", help="IB Gateway/TWS host"),
    port: int = typer.Option(7497, help="IB port (7497 TWS paper, 4002 Gateway paper)"),
    client_id: int = typer.Option(1, help="IB client id"),
    daily_loss_limit: float = typer.Option(0.0, help="Daily-loss kill switch (0 = disabled)"),
    max_position: float = typer.Option(0.0, help="Max abs position (0 = unlimited)"),
    i_understand_live: bool = typer.Option(False, "--i-understand-live",
                                           help="Required to enable LIVE (real money)."),
) -> None:
    """Trade the MA-crossover strategy live on Interactive Brokers (paper by default)."""
    resolved = _resolve_mode(mode, confirm_live=i_understand_live)
    engine = build_live_engine(
        symbol=symbol, mode=resolved, fast=fast, slow=slow, size=size,
        host=host, port=port, client_id=client_id,
        daily_loss_limit=daily_loss_limit or float("inf"),
        max_position=max_position or float("inf"),
    )
    typer.echo(f"connecting to IB {host}:{port} (client {client_id}) in {resolved.value} mode...")
    engine.start()
    typer.echo("connected. running — Ctrl-C to stop.")
    try:
        # Process IB events until interrupted.
        while True:
            engine.broker._client().sleep(1)  # type: ignore[attr-defined]
    except KeyboardInterrupt:
        typer.echo("\nstopping...")
    finally:
        engine.stop()
        typer.echo("stopped; state persisted.")
